## 九、复现方式

```powershell

.\.venv\Scripts\python.exe scripts\project_runner.py setup --config run.audit.json

# 2) 带日志启动（5093）
. .runner\audit\settings\settings.local.ps1
Start-Process .venv\Scripts\python.exe -ArgumentList '-m','flask','--app','app','run','--port','5093','--no-reload' `
  -RedirectStandardError .runner\audit\web.err -RedirectStandardOutput .runner\audit\web.log
Start-Process .venv\Scripts\python.exe -ArgumentList '-m','app.expiry','--interval','60' `
  -RedirectStandardOutput .runner\audit\expiry.log -RedirectStandardError .runner\audit\expiry.err

# 3) 四批脚本
pw.cmd .runner\audit\audit_batch1_rules.py
pw.cmd .runner\audit\audit_batch2_features.py
pw.cmd .runner\audit\audit_batch2b_members_staff.py
```
